CREATE DATABASE IF NOT EXISTS pig;

USE pig;

-- 1. Admin Table
CREATE TABLE IF NOT EXISTS `admin` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `name` varchar(250) DEFAULT NULL,
  `username` varchar(250) DEFAULT NULL,
  `email` varchar(250) DEFAULT NULL,
  `password` varchar(250) DEFAULT NULL
);

INSERT INTO `admin` (`id`, `name`, `username`, `email`, `password`) 
VALUES (1, 'System Admin', 'admin', 'admin@swineguard.com', 'admin')
ON DUPLICATE KEY UPDATE `id`=`id`;


-- 2. Users Table
CREATE TABLE IF NOT EXISTS `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `name` varchar(250) DEFAULT NULL,
  `username` varchar(250) DEFAULT NULL,
  `password` varchar(250) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `email_verified` tinyint(1) NOT NULL DEFAULT 0,
  `verification_code` varchar(10) DEFAULT NULL,
  `code_expires_at` datetime DEFAULT NULL,
  `role` enum('client', 'manager') NOT NULL DEFAULT 'client',
  `status` enum('active', 'disabled') NOT NULL DEFAULT 'active',
  `created_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP
);


-- 3. Sensor Data Table (DHT21 & Water)
CREATE TABLE IF NOT EXISTS `sensor_data` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `temperature` varchar(50) DEFAULT NULL,
  `humidity` varchar(50) DEFAULT NULL,
  `water` varchar(50) DEFAULT NULL,
  `date` varchar(100) DEFAULT NULL,
  `time` varchar(100) DEFAULT NULL
);


-- 4. Machine Real-Time Status Table
CREATE TABLE IF NOT EXISTS `machine_status` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `fan` varchar(50) DEFAULT 'off',
  `exhaust` varchar(50) DEFAULT 'off',
  `water_pump` varchar(50) DEFAULT 'off',
  `heater` varchar(50) DEFAULT 'off',
  `light` varchar(50) DEFAULT 'off'
);


-- 5. System Settings & Bypass Controls Table
CREATE TABLE IF NOT EXISTS `settings_rule` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `temperature_on` varchar(50) DEFAULT '30.0',
  `temperature_off` varchar(50) DEFAULT '26.0',
  `humidity_on` varchar(50) DEFAULT '60.0',
  `humidity_off` varchar(50) DEFAULT '75.0',
  `heater_on` decimal(5,2) DEFAULT 20.00,
  `heater_off` decimal(5,2) DEFAULT 25.00,
  `fan_bypass` enum('AUTO', 'FORCE_ON', 'FORCE_OFF') NOT NULL DEFAULT 'AUTO',
  `exhaust_bypass` enum('AUTO', 'FORCE_ON', 'FORCE_OFF') NOT NULL DEFAULT 'AUTO',
  `water_pump_bypass` enum('AUTO', 'FORCE_ON', 'FORCE_OFF') NOT NULL DEFAULT 'AUTO',
  `heater_bypass` enum('AUTO', 'FORCE_ON', 'FORCE_OFF') NOT NULL DEFAULT 'AUTO',
  `light_bypass` enum('AUTO', 'FORCE_ON', 'FORCE_OFF') NOT NULL DEFAULT 'AUTO'
);


-- 6. State Change Machine Logs Table
CREATE TABLE IF NOT EXISTS `machine_logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `fan` varchar(50) DEFAULT NULL,
  `exhaust` varchar(50) DEFAULT NULL,
  `water_pump` varchar(50) DEFAULT NULL,
  `heater` varchar(50) DEFAULT NULL,
  `light` varchar(50) DEFAULT NULL,
  `date` varchar(100) DEFAULT NULL,
  `time` varchar(100) DEFAULT NULL
);


-- 7. ESP32 Hardware Health & Ping Monitoring Table
CREATE TABLE IF NOT EXISTS `system_health` (
  `id` int(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `device_id` varchar(100) DEFAULT 'ESP32_PIG_GUARD',
  `wifi_signal` varchar(50) DEFAULT NULL,
  `ip_address` varchar(50) DEFAULT NULL,
  `last_ping` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  `status` enum('ONLINE', 'OFFLINE') NOT NULL DEFAULT 'ONLINE'
);
